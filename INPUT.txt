function onInput_AETHER(text) {
  try {
    if (typeof onInput_AETHER_inner === 'function') return onInput_AETHER_inner(text);
    return text;
  } catch (e) {
    try { state.message = "AETHER INPUT: " + (e && e.message ? e.message : e); } catch (e2) {}
    return text;
  }
}

const modifier = (text) => {
  try {
    text = onInput_AETHER(text);
  } catch (e) {
    try { state.message = "AETHER Input: " + (e && e.message ? e.message : e); } catch (e2) {}
  }
  return { text };
};
modifier(text);
