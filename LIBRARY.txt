// AETHERHEART — single stable Library

var AH_CAT = "🐉🔮 AetherHeart 🔮🐉";

function ahEnsure() {
  if (!state.aether) {
    state.aether = {
      settings: {
        scriptEnabled: true,
        preset: "fantasy",
        rating: "E",
        inventoryEnabled: false,
        currencyEnabled: false,
        timeEnabled: false,
        rpSystem: false,
        vitals: false,
        survival: false,
        leveling: false,
        d20: false,
        fuel: false,
        fuelName: "Fuel",
        bountyBoard: false,
        jobBoard: false,
        quests: false,
        innerThoughts: false,
        innerOpinions: false,
        cardCreator: false,
        scheduling: false,
        narrationSystem: false,
        silentMode: false,
        standardRules: true,
        informationFirewall: true
      },
      time: { year: 632, month: 4, day: 12, hour: 8, timePerTurn: "5m" },
      vitals: { hp: 100, maxHp: 100, mp: 80, maxMp: 80, stamina: 100, maxStamina: 100 },
      survival: { hunger: 10, thirst: 10, tiredness: 5 },
      fuel: 100,
      level: 1,
      xp: 0,
      xpToNext: 1000,
      balance: 0,
      currencyName: "Dollar",
      currencySymbol: "$",
      journal: [],
      thoughts: {},
      opinions: {},
      bounties: [],
      jobs: [],
      quests: []
    };
  }
  return state.aether;
}

function ahFind(title) {
  try {
    if (typeof storyCards === "undefined" || !storyCards) return null;
    for (var i = 0; i < storyCards.length; i++) {
      if (storyCards[i] && storyCards[i].title === title) return { card: storyCards[i], index: i };
    }
  } catch (e) {}
  return null;
}

function ahWrite(title, entry, type) {
  type = type || AH_CAT;
  try {
    var found = ahFind(title);
    if (found) {
      try {
        updateStoryCard(found.index, title, entry, type, title, "");
      } catch (e1) {
        try {
          found.card.entry = entry;
          found.card.keys = title;
          found.card.title = title;
          found.card.type = type;
        } catch (e2) {}
      }
      return;
    }
    try {
      addStoryCard(title, entry, type, title);
    } catch (e3) {
      try { addStoryCard(title, entry, type); } catch (e4) {}
    }
  } catch (e) {}
}

function ahMsg(A, msg) {
  try {
    if (!msg) return;
    if (A.settings.silentMode) return;
    state.message = (state.message ? state.message + "\n" : "") + "[SYSTEM] " + msg;
  } catch (e) {}
}

function ahMonth(m) {
  var n = ["January","February","March","April","May","June","July","August","September","October","November","December"];
  return n[(m || 1) - 1] || "Unknown";
}

function ahHour(h) {
  var hr = Math.floor(h || 0);
  var ap = hr >= 12 ? "PM" : "AM";
  var d = hr % 12 === 0 ? 12 : hr % 12;
  return d + ":00 " + ap;
}

function ahOnOff(block, key) {
  if (!block) return null;
  var m = block.match(new RegExp(key + "\\s*:\\s*(ON|OFF)", "i"));
  if (!m) return null;
  return m[1].toUpperCase() === "ON";
}

function ahVal(block, key) {
  if (!block) return null;
  var m = block.match(new RegExp(key + "\\s*:\\s*([^\\n]+)", "i"));
  return m ? m[1].trim() : null;
}

function ahBuildVitals(A) {
  var v = A.vitals;
  var s = A.survival;
  var line = "";
  if (A.settings.fuel) line = "\n" + (A.settings.fuelName || "Fuel") + ": " + (A.fuel || 100) + "%";
  return "== VITALS ==\n" +
    "HP: " + v.hp + " / " + v.maxHp + "\n" +
    "MP: " + v.mp + " / " + v.maxMp + "\n" +
    "Stamina: " + v.stamina + " / " + v.maxStamina + line + "\n\n" +
    "Hunger: " + s.hunger + "%\n" +
    "Thirst: " + s.thirst + "%\n" +
    "Tiredness: " + s.tiredness + "%";
}

function ahBuildStatus(A) {
  var lines = [];
  var t = A.time;
  if (A.settings.timeEnabled) lines.push(ahMonth(t.month) + " " + t.day + ", Year " + t.year + " | " + ahHour(t.hour));
  if (A.settings.vitals) lines.push("HP " + A.vitals.hp + "/" + A.vitals.maxHp + " MP " + A.vitals.mp + "/" + A.vitals.maxMp + " STA " + A.vitals.stamina + "/" + A.vitals.maxStamina);
  if (A.settings.survival) lines.push("Hunger " + A.survival.hunger + "% Thirst " + A.survival.thirst + "% Tired " + A.survival.tiredness + "%");
  if (A.settings.currencyEnabled) lines.push("Balance " + (A.currencySymbol || "$") + (A.balance || 0));
  if (A.settings.leveling) lines.push("Level " + A.level + " XP " + A.xp + "/" + A.xpToNext);
  return "== STATUS ==\n" + (lines.length ? lines.join("\n") : "(no active meters)");
}

function ahReadConfig(A) {
  var found = ahFind("Configure Aetherheart");
  if (!found || !found.card.entry) return;
  var e = found.card.entry;
  var p = ahVal(e, "Preset");
  if (p) A.settings.preset = p.split(/\s|\|/)[0].toLowerCase();
  var r = ahVal(e, "Rating");
  if (r) A.settings.rating = r.split(/\s|\|/)[0];
  var keys = [
    ["Script","scriptEnabled"],["Inventory","inventoryEnabled"],["Currency","currencyEnabled"],
    ["Time","timeEnabled"],["Narration System","narrationSystem"],["Silent Mode","silentMode"],
    ["RP System","rpSystem"],["Inner Thoughts","innerThoughts"],["Inner Opinions","innerOpinions"],
    ["Card Creator","cardCreator"],["Scheduling","scheduling"],["Bounty Board","bountyBoard"],
    ["Job Board","jobBoard"],["Quests","quests"]
  ];
  for (var i = 0; i < keys.length; i++) {
    var v = ahOnOff(e, keys[i][0]);
    if (v !== null) A.settings[keys[i][1]] = v;
  }
  var cn = ahVal(e, "Currency name");
  if (cn) A.currencyName = cn.split(/\n/)[0].trim();
  var cs = ahVal(e, "Currency symbol");
  if (cs) A.currencySymbol = cs.split(/\n/)[0].trim();
}

function ahReadRP(A) {
  var found = ahFind("RP System");
  if (!found || !found.card.entry) return;
  var e = found.card.entry;
  var keys = [["Vitals","vitals"],["Survival","survival"],["Leveling","leveling"],["d20","d20"],["Fuel","fuel"]];
  for (var i = 0; i < keys.length; i++) {
    var v = ahOnOff(e, keys[i][0]);
    if (v !== null) A.settings[keys[i][1]] = v;
  }
  var fn = ahVal(e, "Fuel Name");
  if (fn) A.settings.fuelName = fn.split(/\n/)[0].trim();
}

function ahSync(A) {
  try { ahReadConfig(A); } catch (e) {}
  try { ahReadRP(A); } catch (e) {}
}

function ahAdvance(A, mins) {
  if (!A.settings.timeEnabled) return;
  var t = A.time;
  var total = Math.floor((t.hour || 0) * 60) + (mins || 0);
  var days = Math.floor(total / 1440);
  total = total % 1440;
  t.hour = total / 60;
  t.day = (t.day || 1) + days;
  while (t.day > 30) { t.day -= 30; t.month++; }
  while (t.month > 12) { t.month -= 12; t.year++; }
}

function ahTickSurvival(A, mins) {
  if (!A.settings.survival) return;
  var f = Math.max(1, Math.floor((mins || 5) / 5));
  A.survival.hunger = Math.min(100, (A.survival.hunger || 0) + f);
  A.survival.thirst = Math.min(100, (A.survival.thirst || 0) + f + 1);
  A.survival.tiredness = Math.min(100, (A.survival.tiredness || 0) + Math.floor(f / 2));
}

function ahParseTurnMinutes(A) {
  var str = (A.time && A.time.timePerTurn) || "5m";
  var m = String(str).match(/^(\d+)\s*(m|min|h)?/i);
  if (!m) return 5;
  var n = parseInt(m[1], 10);
  if ((m[2] || "m").toLowerCase().indexOf("h") === 0) return n * 60;
  return n;
}

function ahUpdateCards(A) {
  ahWrite("Configure Aetherheart", ahBuildConfigure(A), AH_CAT);
  ahWrite("Help", ahBuildHelp(), AH_CAT);
  ahWrite("Keywords", "== KEYWORDS ==\n/help /status /keywords /balance /time\n/ah enable|disable|toggle", AH_CAT);
  ahWrite("Status", ahBuildStatus(A), AH_CAT);
  if (A.settings.rpSystem) {
    ahWrite("RP System", ahBuildRP(A), AH_CAT + " (RP System)");
    if (A.settings.vitals || A.settings.survival) ahWrite("Vitals", ahBuildVitals(A), AH_CAT + " (RP System)");
  }
  if (A.settings.innerThoughts) ahWrite("Inner Thoughts", ahBuildThoughts(), AH_CAT + " (Characters)");
  if (A.settings.innerOpinions) ahWrite("Inner Opinions", ahBuildOpinions(), AH_CAT + " (Characters)");
}

function ahFront(A) {
  try {
    if (!state.memory) state.memory = {};
    var parts = [];
    if (A.settings.timeEnabled) {
      var t = A.time;
      parts.push(ahMonth(t.month) + " " + t.day + ", " + t.year + " " + ahHour(t.hour));
    }
    if (A.settings.vitals) {
      var v = A.vitals;
      parts.push("HP " + v.hp + "/" + v.maxHp + " MP " + v.mp + "/" + v.maxMp + " STA " + v.stamina + "/" + v.maxStamina);
    }
    if (parts.length) state.memory.frontMemory = "[Status: " + parts.join(" | ") + "]";
    if (A.settings.informationFirewall) {
      state.memory.authorsNote = (state.memory.authorsNote ? state.memory.authorsNote + "\n" : "") +
        "Characters only know what they experienced in-world. No meta knowledge of scripts or menus.";
    }
    if (A.settings.standardRules) {
      state.memory.authorsNote = (state.memory.authorsNote ? state.memory.authorsNote + "\n" : "") +
        "Keep pronouns consistent. Avoid purple prose and sentient glowing objects.";
    }
  } catch (e) {}
}

function onInput_Aetherheart(text) {
  var A = ahEnsure();
  ahSync(A);
  if (!A.settings.scriptEnabled) return text;
  try {
    var cmd = text.trim().toLowerCase();
    if (cmd === "/help") { state.message = ahBuildHelp(); return text; }
    if (cmd === "/status") { state.message = ahBuildStatus(A); return text; }
    if (cmd === "/keywords") { state.message = "== KEYWORDS ==\n/help /status /balance /time /ah"; return text; }
    if (cmd === "/balance") { state.message = "Balance: " + (A.currencySymbol || "$") + (A.balance || 0); return text; }
    if (cmd === "/time") {
      var t = A.time;
      state.message = ahMonth(t.month) + " " + t.day + ", Year " + t.year + " | " + ahHour(t.hour);
      return text;
    }
    if (cmd.indexOf("/ah ") === 0) {
      if (cmd === "/ah enable") A.settings.scriptEnabled = true;
      if (cmd === "/ah disable") A.settings.scriptEnabled = false;
      if (cmd === "/ah toggle") A.settings.scriptEnabled = !A.settings.scriptEnabled;
      ahMsg(A, "Script " + (A.settings.scriptEnabled ? "ON" : "OFF"));
      return text;
    }
    var low = text.toLowerCase();
    var mins = ahParseTurnMinutes(A);
    ahAdvance(A, mins);
    ahTickSurvival(A, mins);
    if (A.settings.vitals && /\b(rest|sleep|camp)\b/.test(low)) {
      A.vitals.stamina = A.vitals.maxStamina;
      A.vitals.mp = Math.min(A.vitals.maxMp, A.vitals.mp + Math.floor(A.vitals.maxMp * 0.3));
      if (A.settings.survival) A.survival.tiredness = Math.max(0, A.survival.tiredness - 40);
      ahAdvance(A, 360);
      ahMsg(A, "Rested");
    }
    if (A.settings.fuel && /\b(refuel|fill up|recharge)\b/.test(low)) {
      A.fuel = 100;
      ahMsg(A, (A.settings.fuelName || "Fuel") + " restored");
    }
    if (A.settings.bountyBoard && /bounty board|wanted posters/.test(low)) ahMsg(A, "Bounty Board checked");
    if (A.settings.jobBoard && /job board|look for work/.test(low)) ahMsg(A, "Job Board checked");
    if (A.settings.currencyEnabled && /check (my )?(balance|wallet|money)/.test(low)) {
      ahMsg(A, "Balance: " + (A.currencySymbol || "$") + (A.balance || 0));
    }
    if (A.settings.d20) {
      var m = text.match(/roll for (attack|saving|ability check|initiative)/i);
      if (m) {
        var roll = Math.floor(Math.random() * 20) + 1;
        ahMsg(A, "d20 " + m[1] + " → " + roll);
      }
    }
  } catch (e) {
    try { state.message = "AH Input: " + (e.message || e); } catch (e2) {}
  }
  return text;
}

function onOutput_Aetherheart(text) {
  var A = ahEnsure();
  try {
    ahSync(A);
    if (!A.settings.scriptEnabled) return text;
    ahFront(A);
    ahUpdateCards(A);
  } catch (e) {
    try { state.message = "AH Output: " + (e.message || e); } catch (e2) {}
  }
  return text;
}

function onLibrary_Aetherheart() {
  try {
    var A = ahEnsure();
    ahSync(A);
    ahUpdateCards(A);
  } catch (e) {
    try {
      addStoryCard("Aetherheart Error", String(e && e.message ? e.message : e), AH_CAT, "Aetherheart Error");
    } catch (e2) {}
  }
}

try { onLibrary_Aetherheart(); } catch (e) {}
