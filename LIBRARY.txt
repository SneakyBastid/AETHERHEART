// ---------- state ----------

function ahInit() {
  if (!state.AH) {
    state.AH = {
      settings: {
        preset: "fantasy",
        time: true,
        weather: true,
        calendarEvents: true,
        currency: true,
        survival: false,
        vitals: false,
        fuel: false,
        bountyBoard: false,
        jobBoard: false,
        d20: false,
        innerThoughts: false,
        innerOpinions: false,
        cardCreator: false,
        minutesPerTurn: 120,
        startingBalance: 0
      },
      time: { year: 1, month: 1, day: 1, hour: 8, minute: 0, weatherDay: 0, weather: "Clear", season: "Spring" },
      currency: { balance: 0, initialized: false },
      survival: { hunger: 0, thirst: 0, tiredness: 0 },
      vitals: { stamina: 100, maxStamina: 100, mp: 50, maxMp: 50 },
      fuel: 100,
      flags: { rest: false, refuel: false, checkBounty: false, checkJob: false, d20Roll: null },
      message: ""
    };
  }
  // backfill in case new fields get added later
  var d = {
    preset: "fantasy", time: true, weather: true, calendarEvents: true, currency: true,
    survival: false, vitals: false, fuel: false, bountyBoard: false, jobBoard: false,
    d20: false, innerThoughts: false, innerOpinions: false, cardCreator: false,
    minutesPerTurn: 120, startingBalance: 0
  };
  for (var k in d) { if (state.AH.settings[k] === undefined) state.AH.settings[k] = d[k]; }
}

// ---------- story card helpers (real API: addStoryCard(keys, entry, type)) ----------

function ahFindCardIndex(title) {
  for (var i = 0; i < storyCards.length; i++) {
    if (storyCards[i].title === title || storyCards[i].keys === title) return i;
  }
  return -1;
}

// Creates the card if missing, or updates it in place if it exists.
// keys is also what AI Dungeon shows as the card title.
function ahUpsertCard(keys, entry, type) {
  var idx = ahFindCardIndex(keys);
  if (idx === -1) {
    var newIdx = addStoryCard(keys, entry, type || AH_TYPE);
    // addStoryCard returns false if a card with these keys already exists
    if (newIdx === false) {
      idx = ahFindCardIndex(keys);
      if (idx !== -1) updateStoryCard(idx, keys, entry, type || AH_TYPE);
    }
  } else {
    updateStoryCard(idx, keys, entry, type || AH_TYPE);
  }
}

function ahCardEntry(title) {
  var idx = ahFindCardIndex(title);
  if (idx === -1) return null;
  return storyCards[idx].entry;
}

function ahParseBool(line) {
  return /:\s*ON\b/i.test(line);
}
function ahParseNumber(line, fallback) {
  var m = line.match(/:\s*(-?\d+)/);
  return m ? parseInt(m[1], 10) : fallback;
}
function ahParseWord(line, fallback) {
  var m = line.match(/:\s*([a-zA-Z0-9\-]+)/);
  return m ? m[1].toLowerCase() : fallback;
}

function ahLoadConfig() {
  var entry = ahCardEntry(AH_CONFIG_TITLE);
  if (entry === null) {
    // first run: create the card with defaults
    ahUpsertCard(AH_CONFIG_TITLE, ahDefaultConfigText(), AH_TYPE);
    return;
  }
  var lines = entry.split("\n");
  var s = state.AH.settings;
  for (var i = 0; i < lines.length; i++) {
    var l = lines[i];
    if (/^Preset:/i.test(l)) { var p = ahParseWord(l, s.preset); if (AH_PRESETS[p]) s.preset = p; }
    else if (/^Time & Calendar:/i.test(l)) s.time = ahParseBool(l);
    else if (/^Weather:/i.test(l)) s.weather = ahParseBool(l);
    else if (/^Calendar Events:/i.test(l)) s.calendarEvents = ahParseBool(l);
    else if (/^Currency:/i.test(l)) s.currency = ahParseBool(l);
    else if (/^Survival/i.test(l)) s.survival = ahParseBool(l);
    else if (/^Vitals/i.test(l)) s.vitals = ahParseBool(l);
    else if (/^Fuel:/i.test(l)) s.fuel = ahParseBool(l);
    else if (/^Bounty Board:/i.test(l)) s.bountyBoard = ahParseBool(l);
    else if (/^Job Board:/i.test(l)) s.jobBoard = ahParseBool(l);
    else if (/^Dice Rolls/i.test(l)) s.d20 = ahParseBool(l);
    else if (/^Inner Thoughts:/i.test(l)) s.innerThoughts = ahParseBool(l);
    else if (/^Inner Opinions:/i.test(l)) s.innerOpinions = ahParseBool(l);
    else if (/^Card Creator:/i.test(l)) s.cardCreator = ahParseBool(l);
    else if (/^Minutes Per Turn:/i.test(l)) s.minutesPerTurn = ahParseNumber(l, s.minutesPerTurn);
    else if (/^Starting Balance:/i.test(l)) s.startingBalance = ahParseNumber(l, s.startingBalance);
  }
  if (!state.AH.currency.initialized) {
    state.AH.currency.balance = s.startingBalance;
    state.AH.currency.initialized = true;
  }
  ahSyncOptionalCards();
}

// Creates/removes the feature cards to match the toggles, so the
// scenario never ships an empty, unconfigured card.
function ahSyncOptionalCards() {
  var s = state.AH.settings;

  if (s.calendarEvents) {
    if (ahCardEntry(AH_EVENTS_HELP_TITLE) === null) {
      ahUpsertCard(AH_EVENTS_HELP_TITLE,
"Add your own calendar events by creating a card of type " + AH_TYPE + " with:\n\n" +
"Month: 12\n" +
"Day: 25\n" +
"Category: holiday\n" +
"Note: a short line describing what happens on this day.\n\n" +
"Make one card per event. AetherHeart checks every card of this type each turn and " +
"tells the AI when today matches, or when an event is coming up within 7 days. " +
"You do not need to write dates into your scenario text anywhere else.",
        AH_TYPE);
    }
  }

  if (s.innerThoughts) {
    if (ahCardEntry(AH_THOUGHTS_TITLE) === null) {
      ahUpsertCard(AH_THOUGHTS_TITLE,
"Occasionally show the point-of-view character's private inner thoughts in italics, " +
"separate from spoken dialogue. Inner thoughts are honest even when the character's words " +
"are not, and they stay consistent with that character's personality and current situation. " +
"Do not overuse this — a line or two, only when it adds something the reader can't already see.",
        AH_TYPE);
    }
  } else {
    var ti = ahFindCardIndex(AH_THOUGHTS_TITLE);
    if (ti !== -1) removeStoryCard(ti);
  }

  if (s.innerOpinions) {
    if (ahCardEntry(AH_OPINIONS_TITLE) === null) {
      ahUpsertCard(AH_OPINIONS_TITLE,
"Track how the point-of-view character privately feels about the other characters and " +
"events around them, and let those opinions shift as the story goes on. When relevant, " +
"let a character's opinion of someone color how they act toward that person, even if they " +
"never say it out loud.",
        AH_TYPE);
    }
  } else {
    var oi = ahFindCardIndex(AH_OPINIONS_TITLE);
    if (oi !== -1) removeStoryCard(oi);
  }
}

// ---------- currency ----------

function ahCurrencyInfo() {
  return AH_PRESETS[state.AH.settings.preset] || AH_PRESETS.fantasy;
}
function ahFormatBalance() {
  var c = ahCurrencyInfo();
  var bal = state.AH.currency.balance;
  return c.symbol ? (c.symbol + bal) : (bal + " " + c.currency + (bal === 1 ? "" : "s"));
}

// ---------- time & calendar ----------

function ahIsLeap(year) {
  return (year % 4 === 0 && year % 100 !== 0) || (year % 400 === 0);
}
function ahDaysInMonth(month, year) {
  if (month === 2 && ahIsLeap(year)) return 29;
  return AH_DAYS_IN_MONTH[month - 1];
}
function ahSeasonForMonth(month) {
  if (month === 12 || month <= 2) return "Winter";
  if (month <= 5) return "Spring";
  if (month <= 8) return "Summer";
  return "Fall";
}

function ahAdvanceTime(minutes) {
  if (!minutes || minutes <= 0) return;
  var t = state.AH.time;
  t.minute += minutes;
  while (t.minute >= 60) { t.minute -= 60; t.hour++; }
  while (t.hour >= 24) {
    t.hour -= 24;
    t.day++;
  }
  while (t.day > ahDaysInMonth(t.month, t.year)) {
    t.day -= ahDaysInMonth(t.month, t.year);
    t.month++;
    if (t.month > 12) { t.month = 1; t.year++; }
  }
  t.season = ahSeasonForMonth(t.month);

  if (state.AH.settings.weather && t.day !== t.weatherDay) {
    var pool = AH_WEATHER[t.season] || AH_WEATHER.Spring;
    t.weather = pool[Math.floor(Math.random() * pool.length)];
    t.weatherDay = t.day;
  }
}

function ahFormatTime() {
  var t = state.AH.time;
  var h = t.hour % 12; if (h === 0) h = 12;
  var ampm = t.hour < 12 ? "AM" : "PM";
  var m = t.minute < 10 ? "0" + t.minute : t.minute;
  return h + ":" + m + " " + ampm;
}
function ahFormatDate() {
  var t = state.AH.time;
  return AH_MONTHS[t.month - 1] + " " + t.day + ", Year " + t.year;
}

// ---------- calendar events (Realmheart-style: scan cards, don't hardcode) ----------

function ahParseEventCard(card) {
  var e = { month: null, day: null, category: "event", note: "" };
  var lines = card.entry.split("\n");
  for (var i = 0; i < lines.length; i++) {
    var l = lines[i];
    var mm = l.match(/^Month:\s*(\d{1,2})/i);
    var dd = l.match(/^Day:\s*(\d{1,2})/i);
    var cc = l.match(/^Category:\s*(.+)/i);
    var nn = l.match(/^Note:\s*(.+)/i);
    if (mm) e.month = parseInt(mm[1], 10);
    if (dd) e.day = parseInt(dd[1], 10);
    if (cc) e.category = cc[1].trim();
    if (nn) e.note = nn[1].trim();
  }
  if (!e.month || !e.day) return null;
  e.name = card.title || card.keys;
  return e;
}

function ahAllEvents() {
  var out = [];
  for (var i = 0; i < storyCards.length; i++) {
    var c = storyCards[i];
    if (c.type !== AH_TYPE) continue;
    if (c.title === AH_CONFIG_TITLE || c.title === AH_EVENTS_HELP_TITLE ||
        c.title === AH_THOUGHTS_TITLE || c.title === AH_OPINIONS_TITLE) continue;
    var ev = ahParseEventCard(c);
    if (ev) out.push(ev);
  }
  return out;
}

// day-of-year, ignoring leap day for simple wraparound comparisons
function ahDayOfYear(month, day) {
  var total = day;
  for (var m = 1; m < month; m++) total += AH_DAYS_IN_MONTH[m - 1];
  return total;
}

function ahTodayEvents() {
  var t = state.AH.time;
  var all = ahAllEvents();
  var out = [];
  for (var i = 0; i < all.length; i++) {
    if (all[i].month === t.month && all[i].day === t.day) out.push(all[i]);
  }
  return out;
}

function ahUpcomingEvents(withinDays) {
  var t = state.AH.time;
  var todayDoy = ahDayOfYear(t.month, t.day);
  var all = ahAllEvents();
  var out = [];
  for (var i = 0; i < all.length; i++) {
    var doy = ahDayOfYear(all[i].month, all[i].day);
    var diff = doy - todayDoy;
    if (diff < 0) diff += 365;
    if (diff > 0 && diff <= withinDays) out.push(all[i]);
  }
  return out;
}

// ---------- survival / vitals / fuel ----------

function ahTickSurvival(minutes) {
  if (!state.AH.settings.survival) return;
  var s = state.AH.survival;
  var f = Math.max(1, Math.floor((minutes || 5) / 60));
  s.hunger = Math.min(100, s.hunger + f);
  s.thirst = Math.min(100, s.thirst + f + 1);
  s.tiredness = Math.min(100, s.tiredness + Math.floor(f / 2));
}

function ahRest() {
  var v = state.AH.vitals, s = state.AH.survival, set = state.AH.settings;
  if (set.vitals) {
    v.stamina = v.maxStamina;
    v.mp = Math.min(v.maxMp, v.mp + Math.floor(v.maxMp * 0.3));
  }
  if (set.survival) s.tiredness = Math.max(0, s.tiredness - 40);
  ahAdvanceTime(360);
}

function ahRefuel() { state.AH.fuel = 100; }

// ---------- dice ----------

function ahRollD20(text) {
  var m = text.match(/roll for (attack|saving throw|ability check|initiative)/i);
  if (!m) return null;
  var roll = Math.floor(Math.random() * 20) + 1;
  return { kind: m[1], roll: roll };
}

// ---------- card creator ----------
// AI can emit: [NEWCARD: Name | Type | one line description]
// When Card Creator is ON, the script turns that into a real story card
// and strips the tag out of the text the player sees.

function ahRunCardCreator(text) {
  if (!state.AH.settings.cardCreator) return text;
  var re = /\[NEWCARD:\s*([^|]+)\|\s*([^|]+)\|\s*([^\]]+)\]/gi;
  return text.replace(re, function(full, name, type, desc) {
    name = name.trim(); type = type.trim(); desc = desc.trim();
    ahUpsertCard(name, desc, type || "Custom");
    return "";
  });
}

// ---------- the hidden context block ----------

function ahBuildWorldState() {
  var s = state.AH.settings;
  var lines = ["[CURRENT WORLD STATE — treat as ground truth, use to answer any player question about time, date, weather, or balance]"];

  if (s.time) lines.push("Time: " + ahFormatTime());
  if (s.time) lines.push("Date: " + ahFormatDate());
  if (s.weather) lines.push("Weather: " + state.AH.time.weather + " (" + state.AH.time.season + ")");
  if (s.currency) lines.push("Balance: " + ahFormatBalance());

  if (s.calendarEvents) {
    var today = ahTodayEvents();
    if (today.length) {
      var names = [];
      for (var i = 0; i < today.length; i++) names.push(today[i].name + " — " + today[i].note);
      lines.push("Today: " + names.join("; "));
    }
    var soon = ahUpcomingEvents(7);
    if (soon.length) {
      var names2 = [];
      for (var j = 0; j < soon.length; j++) names2.push(soon[j].name);
      lines.push("Upcoming (7 days): " + names2.join(", "));
    }
  }

  if (s.survival) {
    var sv = state.AH.survival;
    lines.push("Hunger: " + sv.hunger + "/100, Thirst: " + sv.thirst + "/100, Tiredness: " + sv.tiredness + "/100");
  }
  if (s.vitals) {
    var v = state.AH.vitals;
    lines.push("Stamina: " + v.stamina + "/" + v.maxStamina + ", MP: " + v.mp + "/" + v.maxMp);
  }
  if (s.fuel) {
    lines.push((ahCurrencyInfo().fuelName || "Fuel") + ": " + state.AH.fuel + "%");
  }

  lines.push("[END WORLD STATE]");
  return lines.join("\n");
}
