function findCard(title) {
  if (!Array.isArray(storyCards)) return null;
  for (let i = 0; i < storyCards.length; i++) {
    if (storyCards[i] && storyCards[i].title === title) return storyCards[i];
  }
  return null;
}

function writeCard(title, entry, category) {
  category = category || "· Aetherheart";
  let card = findCard(title);
  if (!card) {
    try {
      addStoryCard(title, entry, category, title, "{}");
    } catch (e) {}
    return;
  }
  card.entry = entry;
}

function buildBountyBoard(A) {
  let lines = ["== BOUNTY BOARD ==", "Wanted: Dead, Alive, or Either", ""];
  const list = A.bounties || [];
  if (list.length === 0) {
    lines.push("(no active bounties)");
  } else {
    list.forEach((b, i) => {
      lines.push(`${i + 1}. ${b.name} — ${b.condition || "Either"} — ${b.reward || "?"}`);
    });
  }
  lines.push("");
  lines.push("[ADD BOUNTY: Name | Dead/Alive/Either | reward]");
  lines.push("[Natural language: check the bounty board, look at the wanted posters, take the bounty on X, turn in the bounty...]");
  lines.push("[Taking a bounty automatically creates a Quest]");
  return lines.join("\n");
}

function buildJobBoard(A) {
  let lines = ["== JOB BOARD ==", "Available work", ""];
  const list = A.jobs || [];
  if (list.length === 0) {
    lines.push("(no jobs posted)");
  } else {
    list.forEach((j, i) => {
      lines.push(`${i + 1}. ${j.title} — ${j.pay || "?"} — ${j.desc || ""}`);
    });
  }
  lines.push("");
  lines.push("[ADD JOB: Title | pay | description]");
  lines.push("[Natural language: check the job board, look for work, see available jobs, ask about jobs...]");
  return lines.join("\n");
}

function buildStatusCard(A) {
  const t = A.time || {};
  let lines = [];
  lines.push(`${getMonthName(t.month)} ${t.day}, Year ${t.year} | ${formatHour(t.hour)}`);
  if (A.settings.vitals) {
    const v = A.vitals;
    lines.push(`HP ${v.hp}/${v.maxHp}  MP ${v.mp}/${v.maxMp}  STA ${v.stamina}/${v.maxStamina}`);
  }
  if (A.settings.survival) {
    const s = A.survival;
    lines.push(`Hunger ${s.hunger}%  Thirst ${s.thirst}%  Tired ${s.tiredness}%`);
  }
  if (A.settings.fuel) {
    lines.push(`${A.settings.fuelName || "Fuel"}: ${A.fuel || 100}%`);
  }
  if (A.settings.leveling) {
    lines.push(`Level ${A.level}   XP ${A.xp}/${A.xpToNext}`);
  }
  if (A.settings.reputation && A.reputation) {
    lines.push(`Law ${A.reputation.law}  Social ${A.reputation.social}  Heat ${A.reputation.heat || 0}`);
  }
  return `== STATUS ==
${lines.join("\n")}`;
}

function getMonthName(m) {
  return ["January","February","March","April","May","June","July","August","September","October","November","December"][(m || 1) - 1] || "Unknown";
}

function formatHour(h) {
  const hr = Math.floor(h || 0);
  const ampm = hr >= 12 ? "PM" : "AM";
  const d = hr % 12 === 0 ? 12 : hr % 12;
  return d + ":00 " + ampm;
}

function updateRPSystemCards(A) {
  if (!A.settings.rpSystem) return;

  writeCard("RP System", buildRPSystemCard(A), "· Aetherheart (RP System)");

  if (A.settings.vitals) {
    writeCard("Vitals", buildVitalsCard(A), "· Aetherheart (RP System)");
  }
  if (A.settings.leveling) {
    writeCard("Character Sheet", buildCharacterSheet(A), "· Aetherheart (RP System)");
    writeCard("Stat % Modifier", buildStatModifierCard(A), "· Aetherheart (RP System)");
  }
  if (A.settings.bountyBoard) {
    writeCard("Bounty Board", buildBountyBoard(A), "· Aetherheart");
  }
  if (A.settings.jobBoard) {
    writeCard("Job Board", buildJobBoard(A), "· Aetherheart");
  }
  writeCard("Status", buildStatusCard(A), "· Aetherheart");
}

function buildPlayerJournal(A) {
  let lines = ["== PLAYER JOURNAL ==", ""];
  const journal = A.journal || [];
  if (journal.length === 0) {
    lines.push("(empty)");
  } else {
    const recent = journal.slice(-12);
    recent.forEach(entry => lines.push("• " + entry));
  }
  lines.push("");
  lines.push("[Short declarative statements only]");
  lines.push("[Example: You got engaged.]");
  lines.push("[Example: The Dragon of Alsak was defeated by you.]");
  return lines.join("\n");
}

function ensureTitleMarker(entry, name) {
  if (!entry) entry = "";
  if (entry.indexOf("{title:") === -1) {
    entry = "{title: " + name + "}\n" + entry;
  }
  return entry;
}

function buildInventoryCard(A) {
  const inv = A.inventory || {};
  function section(title, obj) {
    const keys = Object.keys(obj || {});
    if (!keys.length) return title + "\n(none)\n";
    return title + "\n" + keys.map(k => k + (obj[k] > 1 ? " x" + obj[k] : "")).join("\n") + "\n";
  }
  return `== INVENTORY ==
${section("-- CLOTHES --", inv.clothes)}
${section("-- ARMOR --", inv.armor)}
${section("-- WEAPONS --", inv.weapons)}
${section("-- AMMO --", inv.ammo)}
${section("-- OTHER --", inv.other)}
Total weight tracked for encumbrance (stat-based).
Armor and weapons default to # (permanent).
[Natural language: give, take, equip, unequip, wear, remove, reload...]
[Brand names used on Modern and Western where possible]`;
}

function buildRouteNetwork(A) {
  const routes = A.routes || {};
  let lines = ["== ROUTE NETWORK ==", "A → B distances (fuel and travel)", ""];
  const keys = Object.keys(routes);
  if (!keys.length) {
    lines.push("(no routes defined)");
  } else {
    keys.forEach(k => lines.push(k + " : " + routes[k]));
  }
  lines.push("");
  lines.push("[ADD ROUTE: PlaceA > PlaceB | distance]");
  lines.push("[Units follow Imperial / Metric toggle]");
  return lines.join("\n");
}

function handleD20(text, A) {
  if (!A.settings.d20) return null;
  const m = text.match(/roll for (attack|saving|ability check(?:\s*\(([^)]+)\))?|initiative)/i);
  if (!m) return null;
  const type = m[1].toLowerCase();
  const roll = Math.floor(Math.random() * 20) + 1;
  let mod = 0;
  // simple stat hook — expand later
  if (type.indexOf("attack") === 0) mod = Math.floor(((A.stats.STR || 10) - 10) / 2);
  if (type.indexOf("ability") === 0) mod = 0;
  const total = roll + mod;
  return `[d20] ${type} → ${roll}${mod ? (mod >= 0 ? "+" : "") + mod : ""} = ${total}`;
}

function handleNaturalLanguage(text, A) {
  const t = text.toLowerCase();
  let messages = [];

  // Bounty Board
  if (A.settings.bountyBoard && isBountyTrigger(text)) {
    messages.push("Bounty Board checked.");
  }
  // Job Board
  if (A.settings.jobBoard && isJobTrigger(text)) {
    messages.push("Job Board checked.");
  }
  // Menu / prices
  if (isMenuTrigger(text)) {
    messages.push("Menu / price list requested.");
  }
  // Equip / inventory natural language
  if (A.settings.inventoryEnabled) {
    if (/\b(equip|wear|put on|don)\b/.test(t)) messages.push("Equip action noted.");
    if (/\b(unequip|remove|take off)\b/.test(t)) messages.push("Unequip action noted.");
    if (/\b(give|hand|offer)\b/.test(t)) messages.push("Give action noted.");
    if (/\b(take|pick up|grab|loot)\b/.test(t)) messages.push("Take action noted.");
    if (/\b(reload)\b/.test(t)) messages.push("Reload action noted.");
  }
  // Rest / heal
  if (A.settings.vitals && /\b(rest|sleep|camp|make camp|take a rest)\b/.test(t)) {
    messages.push("Rest action noted.");
  }
  // Crafting / learning
  if (/\b(craft|forge|brew|cook|build|make)\b/.test(t)) messages.push("Craft action noted.");
  if (/\b(learn|study|read|practice|train)\b/.test(t)) messages.push("Learn action noted.");
  // Fuel
  if (A.settings.fuel && /\b(refuel|fill up|recharge|gas up)\b/.test(t)) {
    messages.push("Fuel action noted.");
  }

  return messages;
}

function applySilentMode(messages, A) {
  if (!messages || !messages.length) return "";
  if (A.settings.silentMode) {

    return ""; // no story text
  }
  return "\n" + messages.map(m => "[SYSTEM] " + m).join("\n");
}

function calcEncumbrance(A) {
  // Simple placeholder: total item count vs CON-based limit
  const inv = A.inventory || {};
  let count = 0;
  ["clothes", "armor", "weapons", "ammo", "other"].forEach(cat => {
    const o = inv[cat] || {};
    Object.keys(o).forEach(k => { count += (o[k] || 1); });
  });
  const limit = 10 + ((A.stats && A.stats.CON) || 10);
  return { count, limit, overloaded: count > limit };
}

function buildEncumbranceLine(A) {
  const e = calcEncumbrance(A);
  return `Encumbrance: ${e.count} / ${e.limit}${e.overloaded ? "  [OVERLOADED]" : ""}`;
}

function encodeSaveState(A) {
  // Compact transferable block (hex-style shorthand)
  try {
    const payload = {
      v: AETHER_VERSION,
      level: A.level,
      xp: A.xp,
      stats: A.stats,
      vitals: A.vitals,
      survival: A.survival,
      balance: A.balance,
      reputation: A.reputation,
      inventory: A.inventory,
      journal: (A.journal || []).slice(-20),
      quests: A.quests,
      time: A.time,
      settings: {
        preset: A.settings.preset,
        rating: A.settings.rating,
        magicMode: A.settings.magicMode
      }
    };
    const json = JSON.stringify(payload);
    // simple base64-ish safe string for copy/paste
    return btoa(unescape(encodeURIComponent(json)));
  } catch (e) {
    return "SAVE_ERROR";
  }
}

function decodeSaveState(block) {
  try {
    const json = decodeURIComponent(escape(atob(block.trim())));
    return JSON.parse(json);
  } catch (e) {
    return null;
  }
}

function applySaveState(A, data) {
  if (!data) return false;
  if (data.level != null) A.level = data.level;
  if (data.xp != null) A.xp = data.xp;
  if (data.stats) A.stats = data.stats;
  if (data.vitals) A.vitals = data.vitals;
  if (data.survival) A.survival = data.survival;
  if (data.balance != null) A.balance = data.balance;
  if (data.reputation) A.reputation = data.reputation;
  if (data.inventory) A.inventory = data.inventory;
  if (data.journal) A.journal = data.journal;
  if (data.quests) A.quests = data.quests;
  if (data.time) A.time = data.time;
  if (data.settings) {
    if (data.settings.preset) A.settings.preset = data.settings.preset;
    if (data.settings.rating) A.settings.rating = data.settings.rating;
    if (data.settings.magicMode) A.settings.magicMode = data.settings.magicMode;
  }
  return true;
}

// Extend command handling
function handleSaveLoadCommands(text, A) {
  const cmd = text.trim();
  if (cmd.toLowerCase() === "/save") {
    const block = encodeSaveState(A);
    state.message = "SAVE STATE (copy this entire block):\n" + block;
    return true;
  }
  if (cmd.toLowerCase().indexOf("/load ") === 0) {
    const block = cmd.slice(6).trim();
    const data = decodeSaveState(block);
    if (applySaveState(A, data)) {
      state.message = "State loaded.";
    } else {
      state.message = "Load failed. Check the block.";
    }
    return true;
  }
  return false;
}

function levelAdjective(level) {
  if (level >= 15) return "Mythic";
  if (level >= 12) return "Legendary";
  if (level >= 9)  return "Epic";
  if (level >= 6)  return "Rare";
  if (level >= 3)  return "Uncommon";
  return "Common";
}

function parseTimePerTurn(str) {
  if (!str) return 5;
  const m = String(str).trim().match(/^(\d+)\s*(m|min|mins|h|hr|hrs)?$/i);
  if (!m) return 5;
  const n = parseInt(m[1], 10);
  const unit = (m[2] || "m").toLowerCase();
  if (unit.indexOf("h") === 0) return n * 60;
  return n; // minutes
}

function advanceTime(A, minutes) {
  if (!A.settings.timeEnabled) return;
  const t = A.time || { year: 632, month: 4, day: 12, hour: 8 };
  let totalMins = Math.floor((t.hour || 0) * 60) + (minutes || 0);
  let dayAdd = Math.floor(totalMins / (24 * 60));
  totalMins = totalMins % (24 * 60);
  t.hour = totalMins / 60;
  t.day = (t.day || 1) + dayAdd;
  while (t.day > 30) { t.day -= 30; t.month++; }
  while (t.month > 12) { t.month -= 12; t.year++; }
  A.time = t;
}

function tickSurvival(A, minutes) {
  if (!A.settings.survival) return;
  const s = A.survival;
  const factor = Math.max(1, Math.floor((minutes || 5) / 5));
  s.hunger = Math.min(100, (s.hunger || 0) + factor);
  s.thirst = Math.min(100, (s.thirst || 0) + factor + 1);
  s.tiredness = Math.min(100, (s.tiredness || 0) + Math.floor(factor / 2));
  if (A.settings.vitals && A.vitals) {
    if (s.hunger > 80) A.vitals.stamina = Math.max(0, A.vitals.stamina - 2);
    if (s.thirst > 80) A.vitals.hp = Math.max(1, A.vitals.hp - 1);
    if (s.tiredness > 85) A.vitals.stamina = Math.max(0, A.vitals.stamina - 3);
  }
}

function applyRest(A) {
  if (!A.settings.vitals) return;
  A.vitals.stamina = A.vitals.maxStamina;
  A.vitals.mp = Math.min(A.vitals.maxMp, A.vitals.mp + Math.floor(A.vitals.maxMp * 0.3));
  if (A.settings.survival) {
    A.survival.tiredness = Math.max(0, A.survival.tiredness - 40);
  }
  advanceTime(A, 60 * 6); // 6 hours
}

function inventoryAdd(A, category, name, qty) {
  if (!A.inventory) A.inventory = { clothes: {}, armor: {}, weapons: {}, ammo: {}, other: {} };
  if (!A.inventory[category]) A.inventory[category] = {};
  const q = qty || 1;
  A.inventory[category][name] = (A.inventory[category][name] || 0) + q;
}

function inventoryRemove(A, category, name, qty) {
  if (!A.inventory || !A.inventory[category]) return false;
  const have = A.inventory[category][name] || 0;
  const q = qty || 1;
  if (have < q) return false;
  A.inventory[category][name] = have - q;
  if (A.inventory[category][name] <= 0) delete A.inventory[category][name];
  return true;
}

function handleInventoryVerbs(text, A) {
  if (!A.settings.inventoryEnabled) return [];
  const t = text.toLowerCase();
  const msgs = [];

  // very light pattern grabbers — expand with better NLP later
  let m;
  m = t.match(/\b(?:pick up|take|grab|loot)\s+(?:the\s+)?([a-z0-9\s\-']{2,40})/);
  if (m) {
    inventoryAdd(A, "other", m[1].trim(), 1);
    msgs.push("Took " + m[1].trim());
  }
  m = t.match(/\b(?:equip|wear|put on|don)\s+(?:the\s+)?([a-z0-9\s\-']{2,40})/);
  if (m) {
    msgs.push("Equipped " + m[1].trim());
  }
  m = t.match(/\b(?:unequip|remove|take off)\s+(?:the\s+)?([a-z0-9\s\-']{2,40})/);
  if (m) {
    msgs.push("Removed " + m[1].trim());
  }
  m = t.match(/\breload\b/);
  if (m) msgs.push("Reloaded");

  return msgs;
}

function handleBountyActions(text, A) {
  if (!A.settings.bountyBoard) return [];
  const t = text.toLowerCase();
  const msgs = [];

  let m = t.match(/\btake (?:the )?bounty on ([a-z0-9\s\-']{2,40})/);
  if (m) {
    const name = m[1].trim();
    A.bounties = A.bounties || [];
    A.bounties.push({ name: name, condition: "Either", reward: "TBD", taken: true });
    // auto quest
    A.quests = A.quests || [];
    if (A.quests.length < 5) {
      A.quests.push({ title: "Bounty: " + name, progress: 0 });
    }
    msgs.push("Bounty taken on " + name + " (quest added)");
  }

  m = t.match(/\bturn in (?:the )?bounty on ([a-z0-9\s\-']{2,40})/);
  if (m) {
    const name = m[1].trim();
    msgs.push("Bounty turned in: " + name);
    // simple reward hook
    if (A.settings.currencyEnabled) A.balance = (A.balance || 0) + 50;
  }

  return msgs;
}

function handleJobActions(text, A) {
  if (!A.settings.jobBoard) return [];
  const t = text.toLowerCase();
  const msgs = [];
  if (/accept (?:the )?job|take (?:the )?job|i'?ll take (?:that|the) job/.test(t)) {
    msgs.push("Job accepted");
  }
  if (/finish (?:the )?job|complete (?:the )?job|turn in (?:the )?job/.test(t)) {
    msgs.push("Job completed");
    if (A.settings.currencyEnabled) A.balance = (A.balance || 0) + 25;
  }
  return msgs;
}

function handleRestAndFuel(text, A) {
  const t = text.toLowerCase();
  const msgs = [];
  if (A.settings.vitals && /\b(rest|sleep|camp|make camp|take a rest)\b/.test(t)) {
    applyRest(A);
    msgs.push("Rested");
  }
  if (A.settings.fuel && /\b(refuel|fill up|recharge|gas up)\b/.test(t)) {
    A.fuel = 100;
    msgs.push((A.settings.fuelName || "Fuel") + " restored");
  }
  return msgs;
}

function processTurnSystems(text, A) {
  const minutes = parseTimePerTurn(A.time && A.time.timePerTurn);
  advanceTime(A, minutes);
  tickSurvival(A, minutes);

  let msgs = [];
  msgs = msgs.concat(handleInventoryVerbs(text, A));
  msgs = msgs.concat(handleBountyActions(text, A));
  msgs = msgs.concat(handleJobActions(text, A));
  msgs = msgs.concat(handleRestAndFuel(text, A));

  if (msgs.length && !A.settings.silentMode) {
    state.message = (state.message ? state.message + "\n" : "") + msgs.map(m => "[SYSTEM] " + m).join("\n");
  }
}

function ensureCharacterMemory(A, name) {
  if (!A.charMemory) A.charMemory = {};
  if (!A.charMemory[name]) {
    A.charMemory[name] = {
      lastState: "",
      goal: "",
      mood: "",
      trust: 0,
      debt: 0,
      secret: "",
      remembersAboutPlayer: []
    };
  }
  return A.charMemory[name];
}

function buildCharacterMemoryCard(A) {
  let lines = ["== CHARACTER MEMORY ==", "Short per-character state", ""];
  const cm = A.charMemory || {};
  const names = Object.keys(cm);
  if (!names.length) {
    lines.push("(none tracked)");
  } else {
    names.slice(0, 8).forEach(n => {
      const c = cm[n];
      lines.push(n);
      lines.push("  State: " + (c.lastState || "-"));
      lines.push("  Goal: " + (c.goal || "-"));
      lines.push("  Mood: " + (c.mood || "-"));
      lines.push("  Trust: " + c.trust + "  Debt: " + c.debt);
      if (c.remembersAboutPlayer && c.remembersAboutPlayer.length) {
        lines.push("  Remembers: " + c.remembersAboutPlayer.slice(0, 5).join("; "));
      }
      lines.push("");
    });
  }
  lines.push("[Max a few sentences per person]");
  lines.push("[Compresses when the list grows too long — keywords kept]");
  return lines.join("\n");
}

function compressThoughts(A) {
  // Keep keywords, shorten long notes lists
  if (!A.thoughts) return;
  const keys = Object.keys(A.thoughts);
  if (keys.length <= 12) return;
  const keep = keys.slice(-10);
  const next = {};
  keep.forEach(k => { next[k] = A.thoughts[k]; });
  A.thoughts = next;
}

function expandNaturalLanguage(text, A) {
  const t = text.toLowerCase();
  const msgs = [];

  // Time / schedule
  if (/\bwhat time is it\b|\bcheck the time\b|\bcurrent time\b/.test(t)) {
    msgs.push("Time: " + formatHour((A.time || {}).hour));
  }
  if (/\bwhat('?s| is) (on )?the schedule\b|\bcheck (the )?schedule\b/.test(t)) {
    msgs.push("Schedule checked");
  }

  // Reputation / heat
  if (A.settings.reputation && /\bcheck (my )?reputation\b|\bhow (am i|do i) stand\b|\bwanted level\b/.test(t)) {
    const r = A.reputation || {};
    msgs.push("Law " + (r.law || 0) + " | Social " + (r.social || 0) + " | Heat " + (r.heat || 0));
  }

  // Quest log
  if (A.settings.quests && /\bcheck (my )?quests\b|\bquest log\b|\bwhat am i doing\b/.test(t)) {
    msgs.push("Quests: " + ((A.quests || []).length));
  }

  // Sleeve / shift / isekai flavor triggers
  if (A.settings.sleeveJump && /\b(resleeve|new sleeve|sleeve jump|change bodies)\b/.test(t)) {
    msgs.push("Sleeve jump requested");
  }
  if (A.settings.shift && /\b(time shift|shift (back|forward)|jump (back|forward) in time)\b/.test(t)) {
    msgs.push("Shift requested");
  }
  if (A.settings.isekai && /\b(isekai|transported|summoned to another world)\b/.test(t)) {
    msgs.push("Isekai displacement flag");
  }

  // Deus Ex intentional
  if (A.settings.deusEx && /\b(god mode|deus ex|make me op)\b/.test(t)) {
    msgs.push("Deus Ex acknowledged");
  }

  return msgs;
}

function filterBannedPhrases(text) {
  if (!text) return text;
  let out = text;
  // Soft-clean common annoyances
  out = out.replace(/[“”]/g, '"');
  out = out.replace(/\s*—\s*/g, " - ");
  out = out.replace(/\.{4,}/g, "...");
  // Reduce obvious spam patterns (light touch)
  out = out.replace(/(psychic aftershocks?\.)(\s*\1)+/gi, "$1");
  out = out.replace(/(the (?:pendant|amulet|crystal) pulsed[^.]*\.)(\s*\1)+/gi, "$1");
  return out;
}

function parseOnOff(block, key) {
  if (!block) return null;
  const re = new RegExp(key + "\\s*:\\s*(ON|OFF)", "i");
  const m = block.match(re);
  if (!m) return null;
  return m[1].toUpperCase() === "ON";
}

function parseValue(block, key) {
  if (!block) return null;
  const re = new RegExp(key + "\\s*:\\s*([^\\n]+)", "i");
  const m = block.match(re);
  return m ? m[1].trim() : null;
}

function readConfigureCard(A) {

  if (preset) {
    const p = preset.split(/\s|\|/)[0].toLowerCase();
    if (p) A.settings.preset = p;
  }
  const rating = parseValue(e, "Rating");
  if (rating) A.settings.rating = rating.split(/\s|\|/)[0];

  // Core toggles
  const map = [

function handleAddCommands(text, A) {
  const lines = text.split(/\n/);
  let msgs = [];
  lines.forEach(line => {
    let m;

    // Skills / Abilities / Powers
    m = line.match(/\[ADD SKILL:\s*([^\]]+)\]/i);
    if (m) {
      A.skills = A.skills || [];
      A.skills.push(m[1].trim());
      msgs.push("Skill added: " + m[1].trim());
    }
    m = line.match(/\[ADD ABILITY:\s*([^\]]+)\]/i);
    if (m) {
      A.abilities = A.abilities || [];
      A.abilities.push(m[1].trim());
      msgs.push("Ability added: " + m[1].trim());
    }
    m = line.match(/\[ADD POWER:\s*([^\]]+)\]/i);
    if (m) {
      A.powers = A.powers || [];
      A.powers.push(m[1].trim());
      msgs.push("Power added: " + m[1].trim());
    }

    // Bounty
    m = line.match(/\[ADD BOUNTY:\s*([^\]|]+)\|\s*([^\]|]+)\|\s*([^\]]+)\]/i);
    if (m) {
      A.bounties = A.bounties || [];
      A.bounties.push({
        name: m[1].trim(),
        condition: m[2].trim(),
        reward: m[3].trim()
      });
      msgs.push("Bounty posted: " + m[1].trim());
    }

    // Job
    m = line.match(/\[ADD JOB:\s*([^\]|]+)\|\s*([^\]|]+)\|\s*([^\]]+)\]/i);
    if (m) {
      A.jobs = A.jobs || [];
      A.jobs.push({
        title: m[1].trim(),
        pay: m[2].trim(),
        desc: m[3].trim()
      });
      msgs.push("Job posted: " + m[1].trim());
    }

    // Quest
    m = line.match(/\[ADD QUEST:\s*([^\]|]+)(?:\|\s*([^\]]+))?\]/i);
    if (m) {
      A.quests = A.quests || [];
      if (A.quests.length < 5) {
        A.quests.push({ title: m[1].trim(), progress: 0, desc: (m[2] || "").trim() });
        msgs.push("Quest added: " + m[1].trim());
      }
    }

    // Route
    m = line.match(/\[ADD ROUTE:\s*([^\]|]+)\|\s*([^\]]+)\]/i);
    if (m) {
      A.routes = A.routes || {};
      A.routes[m[1].trim()] = m[2].trim();
      msgs.push("Route added: " + m[1].trim());
    }

    // Schedule block
    m = line.match(/\[ADD:\s*([^\]|]+)\|\s*([^\]|]+)\|\s*([^\]|]+)\|\s*([^\]|]+)\|\s*([^\]]+)\]/i);
    if (m) {
      A.schedule = A.schedule || {};
      const key = m[2].trim();
      if (!A.schedule[key]) A.schedule[key] = [];
      A.schedule[key].push({
        name: m[1].trim(),
        start: m[3].trim(),
        end: m[4].trim(),
        desc: m[5].trim(),
        on: true
      });
      msgs.push("Schedule block added: " + m[1].trim());
    }
  });
  return msgs;
}

function detectCardType(details) {
  const t = (details || "").toLowerCase();
  if (/location|town|city|shop|tavern|room|building|street/.test(t)) return "Location";
  if (/person|man|woman|character|npc|ally|enemy|girl|boy/.test(t)) return "Character";
  return "Other";
}

function buildGeneratedEntry(type, title, details, A) {
  let entry = "";
  if (A.settings.innerThoughts) {
    entry += "{title: " + title + "}\n";
  }
  if (type === "Character") {
    entry += "Name: " + title + "\n";
    entry += "Appearance: concise physical description pending\n";
    entry += "Personality: \n";
    entry += "Background: \n";
    if (details) entry += "Notes: " + details + "\n";
  } else if (type === "Location") {
    entry += "Name: " + title + "\n";
    entry += "Description: " + (details || "") + "\n";
  } else {
    entry += "Name: " + title + "\n";
    entry += "Description: " + (details || "") + "\n";
  }
  return entry;
}

function handleCardCommand(text, A) {
  // /card (type) (title) (details)
  const m = text.match(/^\/card\s+(character|location|other)?\s*([^\n]+)$/i);
  if (!m) return false;

  let type = (m[1] || "").trim();
  let rest = m[2].trim();
  // title is first chunk, details after
  let title = rest;
  let details = "";
  const sp = rest.indexOf(" ");
  if (sp > 0) {
    // allow: /card Character Bob tall scarred man
    const parts = rest.split(/\s+/);
    if (!type) {
      // try first word as type
      const maybe = parts[0].toLowerCase();
      if (maybe === "character" || maybe === "location" || maybe === "other") {
        type = maybe;
        parts.shift();
      }
    }
    title = parts[0];
    details = parts.slice(1).join(" ");
  }
  if (!type) type = detectCardType(details || title);
  type = type.charAt(0).toUpperCase() + type.slice(1).toLowerCase();

  const entry = buildGeneratedEntry(type, title, details, A);
  const category = "· Aetherheart (Card Creator)";
  writeCard(title, entry, category);

  if (!A.settings.silentMode) {
    state.message = "[SYSTEM] Card created: " + title + " (" + type + ")";
  }
  // Suspend normal story continuation feel
  return true;
}

function updateMemoryNotes(A, name, sentence) {
  if (!A.settings.innerThoughts) return;
  A.thoughts = A.thoughts || {};
  if (!A.thoughts[name]) A.thoughts[name] = "";
  // append short declarative memory
  A.thoughts[name] = (A.thoughts[name] + " " + sentence).trim();
  // keep relatively short
  if (A.thoughts[name].length > 400) {
    A.thoughts[name] = A.thoughts[name].slice(-350);
  }
  // write into the card notes if the card exists
  const card = findCard(name);
  if (card) {
    let entry = card.entry || "";
    if (entry.indexOf("{title:") === -1 && A.settings.innerThoughts) {
      entry = "{title: " + name + "}\n" + entry;
    }
    // replace or append Notes line
    if (/Notes:/i.test(entry)) {
      entry = entry.replace(/Notes:.*$/im, "Notes: " + A.thoughts[name]);
    } else {
      entry = entry + "\nNotes: " + A.thoughts[name];
    }
    card.entry = entry;
  }
}

function ledgerAdd(A, note, amount) {

function handleCurrencySpeech(text, A) {
  if (!A.settings.currencyEnabled) return [];
  const t = text.toLowerCase();
  const msgs = [];

  // simple pay / buy patterns
  let m = t.match(/\b(?:pay|spend|tip)\s+(\d+)/);
  if (m) {
    const amt = parseInt(m[1], 10);
    if ((A.balance || 0) >= amt) {
      ledgerAdd(A, "Payment", -amt);
      msgs.push("Paid " + amt);
    } else {
      msgs.push("Insufficient funds");
    }
  }

  m = t.match(/\b(?:buy|purchase)\s+(?:a |an |the )?([a-z0-9\s\-']{2,40})(?:\s+for\s+(\d+))?/);
  if (m) {
    const item = m[1].trim();
    const price = m[2] ? parseInt(m[2], 10) : 10;
    if ((A.balance || 0) >= price) {
      ledgerAdd(A, "Bought " + item, -price);
      inventoryAdd(A, "other", item, 1);
      msgs.push("Bought " + item + " for " + price);
    } else {
      msgs.push("Cannot afford " + item);
    }
  }

  m = t.match(/\b(?:receive|earn|get paid|collect)\s+(\d+)/);
  if (m) {
    const amt = parseInt(m[1], 10);
    ledgerAdd(A, "Income", amt);
    msgs.push("Received " + amt);
  }

  if (/\bcheck (my )?(balance|wallet|money)\b/.test(t)) {
    msgs.push("Balance: " + (A.balance || 0));
  }

  return msgs;
}

function gainXP(A, amount) {
  if (!A.settings.leveling) return;
  A.xp = (A.xp || 0) + (amount || 0);
  while (A.xp >= (A.xpToNext || 1000)) {
    A.xp -= A.xpToNext;
    A.level = (A.level || 1) + 1;
    A.xpToNext = Math.floor((A.xpToNext || 1000) * 1.35);
    // small vitals bump
    if (A.vitals) {
      A.vitals.maxHp += 5;
      A.vitals.hp = A.vitals.maxHp;
      A.vitals.maxStamina += 3;
      A.vitals.stamina = A.vitals.maxStamina;
    }
    if (!A.settings.silentMode) {
      state.message = (state.message ? state.message + "\n" : "") +
        "[SYSTEM] Level up! Now level " + A.level;
    }
    // level loot flavor name available via generateLevelLoot
  }
}

function handleXPSpeech(text, A) {
  if (!A.settings.leveling) return [];

}

function handleDrugUse(text, A) {
  const t = text.toLowerCase();
  const msgs = [];
  // mirrors Configure card defaults; expand with parsed drug list later
  if (/\b(smoke|hit|use)\b.*\b(blunt|joint|weed)\b/.test(t) ||
      /\b(blunt|joint|weed)\b.*\b(smoke|hit|use)\b/.test(t)) {
    if (A.settings.survival) {
      A.survival.hunger = Math.min(100, (A.survival.hunger || 0) + 20);
      A.survival.thirst = Math.min(100, (A.survival.thirst || 0) + 40);
      A.survival.tiredness = Math.min(100, (A.survival.tiredness || 0) + 30);
    }
    if (A.settings.vitals && A.vitals) {

    }
    msgs.push("Used weed");

  return msgs;
}

function parseLocationHours(entry) {
  // {hours: 6:00 AM | 5:00 PM} or hours: ...
  if (!entry) return null;
  let m = entry.match(/\{hours:\s*([^|]+)\|\s*([^}]+)\}/i);
  if (!m) m = entry.match(/hours:\s*([^|]+)\|\s*([^\n]+)/i);
  if (!m) return null;
  return { open: m[1].trim(), close: m[2].trim() };
}

function parseCharacterShift(entry) {
  // {shift: Tavern | 8:00 PM | 2:00AM}
  if (!entry) return null;
  const m = entry.match(/\{shift:\s*([^|]+)\|\s*([^|]+)\|\s*([^}]+)\}/i);
  if (!m) return null;
  return { place: m[1].trim(), start: m[2].trim(), end: m[3].trim() };
}

function scanLocationAndShiftCards(A) {
  if (!Array.isArray(storyCards)) return;
  A.locationHours = A.locationHours || {};
  A.characterShifts = A.characterShifts || {};
  storyCards.forEach(c => {
    if (!c || !c.entry) return;
    const hours = parseLocationHours(c.entry);
    if (hours) A.locationHours[c.title] = hours;
    const shift = parseCharacterShift(c.entry);
    if (shift) A.characterShifts[c.title] = shift;
  });
}

function buildLocationShiftInfoCard(A) {
  let lines = ["== HOURS & SHIFTS ==", ""];
  const lh = A.locationHours || {};
  const cs = A.characterShifts || {};
  const locKeys = Object.keys(lh);
  const shiftKeys = Object.keys(cs);
  if (!locKeys.length && !shiftKeys.length) {
    lines.push("(none detected)");
  } else {
    locKeys.forEach(k => lines.push(k + " hours: " + lh[k].open + " - " + lh[k].close));
    shiftKeys.forEach(k => lines.push(k + " shift: " + cs[k].place + " " + cs[k].start + "-" + cs[k].end));
  }
  lines.push("");
  lines.push("[Put {hours: open | close} in location cards]");
  lines.push("[Put {shift: Place | start | end} in character cards]");
  return lines.join("\n");
}

function isNight(A) {
  const h = (A.time && A.time.hour) || 0;
  return h >= 20 || h < 6;
}

function handleTransformation(text, A) {
  if (!A.settings.rpSystem) return [];
  const t = ensureTransformConfig(A);
  const msgs = [];
  const lower = text.toLowerCase();

  // anger trigger
  if (t.anger && /\b(rage|fury|enraged|lose control|berserk)\b/.test(lower)) {
    t.form = "beast";
    msgs.push("Transformation triggered by anger");
  }

  // time-based / every night
  if (t.timeBased && t.everyNight && isNight(A) && t.form === "human") {
    t.form = "beast";
    msgs.push("Night transformation");
  }
  if (t.timeBased && !isNight(A) && t.form === "beast" && t.daylight === false) {
    // stays beast if cannot be in daylight as human — simplified
  }
  if (!isNight(A) && t.form === "beast" && t.daylight) {
    // optional auto-revert at dawn
    // t.form = "human";
  }

  // feeding
  if (/\b(feed|feeding|drink blood|hunt)\b/.test(lower) && t.form !== "human") {
    if (t.feedingRestores === "HP" && A.vitals) {
      A.vitals.hp = Math.min(A.vitals.maxHp, A.vitals.hp + 20);
      msgs.push("Feeding restored HP");
    }
  }

  // mer flavor
  if (/\b(enter the water|dive|swim out)\b/.test(lower) && /\bmer\b|mermaid|merman/.test(lower)) {
    t.form = "mer";
    msgs.push("Mer form");
  }

  return msgs;
}

function handleScheduleDel(text, A) {
  const msgs = [];
  let m = text.match(/\[DEL:\s*([^\]]+)\]/i);
  if (!m) return msgs;
  const target = m[1].trim();
  // delete whole day
  if (A.schedule && A.schedule[target]) {
    delete A.schedule[target];
    msgs.push("Removed schedule day: " + target);
    return msgs;
  }
  // delete block by name
  const sched = A.schedule || {};
  Object.keys(sched).forEach(day => {
    const before = sched[day].length;
    sched[day] = sched[day].filter(b => b.name.toLowerCase() !== target.toLowerCase());
    if (sched[day].length !== before) msgs.push("Removed block: " + target);
  });
  return msgs;
}

function ensureInjuries(A) {
  if (!A.injuries) A.injuries = [];
  if (!A.scars) A.scars = [];
  return A;
}

function addInjury(A, name, severity) {
  ensureInjuries(A);
  A.injuries.push({ name: name, severity: severity || 1, turns: 0 });
}

function tickInjuries(A) {
  if (!A.settings.vitals) return;
  ensureInjuries(A);
  const keep = [];
  A.injuries.forEach(inj => {
    inj.turns = (inj.turns || 0) + 1;
    // fade: low severity clears faster
    const fadeAfter = 8 - Math.min(5, inj.severity || 1);
    if (inj.turns >= fadeAfter) {
      // chance to leave a scar
      if ((inj.severity || 1) >= 3) {
        A.scars.push(inj.name);
      }
    } else {
      keep.push(inj);
    }
  });
  A.injuries = keep;
}

function buildInjuriesCard(A) {
  ensureInjuries(A);
  let lines = ["== INJURIES & SCARS ==", ""];
  if (!A.injuries.length) lines.push("Injuries: (none)");
  else {
    lines.push("Injuries:");
    A.injuries.forEach(i => lines.push("- " + i.name + " (sev " + i.severity + ")"));
  }
  lines.push("");
  if (!A.scars.length) lines.push("Scars: (none)");
  else {
    lines.push("Scars:");
    A.scars.forEach(s => lines.push("- " + s));
  }
  lines.push("");
  lines.push("[Injuries fade over turns. Severe ones can leave scars.]");
  return lines.join("\n");
}

function tickHeat(A) {
  if (!A.settings.reputation) return;
  A.reputation = A.reputation || { law: 0, social: 0, bounty: 0, heat: 0 };
  if (A.reputation.heat > 0) {
    // slow natural decay
    A.reputation.heat = Math.max(0, A.reputation.heat - 1);
  }
}

function startJail(A, hours) {
  if (!A.settings.jail || !A.settings.timeEnabled) return;
  A.jail = { remaining: hours || 24, active: true };
  if (!A.settings.silentMode) {
    state.message = (state.message ? state.message + "\n" : "") +
      "[SYSTEM] Jailed for " + (hours || 24) + " hours.";
  }
}

function tickJail(A, minutes) {
  if (!A.jail || !A.jail.active) return;
  A.jail.remaining = Math.max(0, (A.jail.remaining || 0) - ((minutes || 5) / 60));
  if (A.jail.remaining <= 0) {
    A.jail.active = false;
    if (!A.settings.silentMode) {
      state.message = (state.message ? state.message + "\n" : "") +
        "[SYSTEM] Released from jail.";
    }
  }
}

function handleInjurySpeech(text, A) {
  if (!A.settings.vitals) return [];
  const t = text.toLowerCase();
  const msgs = [];
  if (/\b(wound|injured|bleeding|broken arm|broken leg|cut|stabbed|shot)\b/.test(t)) {
    let name = "Wound";
    if (/broken arm/.test(t)) name = "Broken arm";
    else if (/broken leg/.test(t)) name = "Broken leg";
    else if (/shot/.test(t)) name = "Gunshot";
    else if (/stabbed/.test(t)) name = "Stab wound";
    addInjury(A, name, /broken|shot|stabbed/.test(t) ? 3 : 1);
    msgs.push("Injury recorded: " + name);
  }
  return msgs;
}

function handleJailSpeech(text, A) {
  if (!A.settings.jail) return [];
  const t = text.toLowerCase();
  const msgs = [];
  if (/\b(arrested|thrown in jail|locked up|sent to jail)\b/.test(t)) {
    startJail(A, 24);
    msgs.push("Jail started");
  }

function handleSleeveJump(text, A) {
  if (!A.settings.sleeveJump) return [];
  const t = text.toLowerCase();
  const msgs = [];
  if (!/\b(resleeve|new sleeve|sleeve jump|change bodies|download into)\b/.test(t)) return msgs;

  ensureJumpState(A);
  // Keep mental stats, replace physical
  const mental = {
    INT: A.stats.INT,
    WIS: A.stats.WIS,
    CHA: A.stats.CHA
  };
  // Physical reroll baseline (simple)
  A.stats.STR = 8 + Math.floor(Math.random() * 7);
  A.stats.DEX = 8 + Math.floor(Math.random() * 7);
  A.stats.CON = 8 + Math.floor(Math.random() * 7);
  A.stats.INT = mental.INT;
  A.stats.WIS = mental.WIS;
  A.stats.CHA = mental.CHA;
  if (A.vitals) {
    A.vitals.maxHp = 80 + A.stats.CON * 2;
    A.vitals.hp = A.vitals.maxHp;
    A.vitals.maxStamina = 80 + A.stats.CON;
    A.vitals.stamina = A.vitals.maxStamina;
  }
  A.sleeve = { at: (A.time ? A.time.year + "-" + A.time.month + "-" + A.time.day : "now") };
  msgs.push("Sleeve jump complete — physical stats replaced, mental kept");
  return msgs;
}

function handleShift(text, A) {
  if (!A.settings.shift) return [];
  const t = text.toLowerCase();

  ensureJumpState(A);

  if (/\b(time shift|shift (back|forward)|jump (back|forward) in time)\b/.test(t)) {
    if (!A.shiftState.active) {
      A.shiftState.active = true;
      A.shiftState.origin = JSON.parse(JSON.stringify(A.time || {}));
      A.shiftState.changes = [];
      msgs.push("Shift started — actions may alter the present on return");
    }
  }
  if (A.shiftState.active && /\b(return to (the )?present|end shift|shift back home)\b/.test(t)) {
    A.shiftState.active = false;
    msgs.push("Returned to main line — timeline updates applied");
    // changes list can drive narrative; state already mutated live

  return msgs;
}

function ensureEquip(A) {
  if (!A.equipped) {
    A.equipped = { weapon: null, armor: null, clothes: null };
  }
  if (!A.inventory) {
    A.inventory = { clothes: {}, armor: {}, weapons: {}, ammo: {}, other: {} };
  }
  return A;

function handleAmmo(text, A) {
  if (!A.settings.ammo) return [];
  const t = text.toLowerCase();
  const msgs = [];
  ensureEquip(A);

  if (/\b(reload|reload my|load rounds|load shells)\b/.test(t)) {
    const ammoKeys = Object.keys(A.inventory.ammo || {});
    if (ammoKeys.length) {
      const k = ammoKeys[0];
      if ((A.inventory.ammo[k] || 0) > 0) {
        A.inventory.ammo[k] -= 1;
        if (A.inventory.ammo[k] <= 0) delete A.inventory.ammo[k];
        msgs.push("Reloaded using " + k);
      } else msgs.push("No ammo left");
    } else msgs.push("No ammo in inventory");
  }

  if (/\b(shoot|fire|gunshot|empty the clip)\b/.test(t)) {
    const ammoKeys = Object.keys(A.inventory.ammo || {});
    if (ammoKeys.length && (A.inventory.ammo[ammoKeys[0]] || 0) > 0) {
      A.inventory.ammo[ammoKeys[0]] -= 1;
      if (A.inventory.ammo[ammoKeys[0]] <= 0) delete A.inventory.ammo[ammoKeys[0]];
      tickMomentum(A, true);
      msgs.push("Shot fired");
    } else {
      tickMomentum(A, false);
      msgs.push("Click — empty");
    }
  }
  return msgs;
}

function handleEquipSlots(text, A) {
  if (!A.settings.inventoryEnabled) return [];
  const t = text.toLowerCase();
  const msgs = [];
  ensureEquip(A);

  let m = t.match(/\b(?:equip|wear|put on|don)\s+(?:the\s+)?([a-z0-9\s\-']{2,40})/);
  if (m) {
    const item = m[1].trim();
    // naive slot guess
    if (/(armor|vest|mail|plate|helmet|pads)/.test(item)) {
      A.equipped.armor = item;
      msgs.push("Armor equipped: " + item);
    } else if (/(sword|gun|rifle|pistol|bow|knife|blade|staff|hammer|bat)/.test(item)) {
      A.equipped.weapon = item;
      msgs.push("Weapon equipped: " + item);
    } else {
      A.equipped.clothes = item;
      msgs.push("Wearing: " + item);
    }
  }

  m = t.match(/\b(?:unequip|remove|take off)\s+(?:the\s+)?([a-z0-9\s\-']{2,40})/);
  if (m) {
    const item = m[1].trim();
    if (A.equipped.weapon && A.equipped.weapon.indexOf(item) !== -1) A.equipped.weapon = null;
    if (A.equipped.armor && A.equipped.armor.indexOf(item) !== -1) A.equipped.armor = null;
    if (A.equipped.clothes && A.equipped.clothes.indexOf(item) !== -1) A.equipped.clothes = null;
    msgs.push("Removed: " + item);
  }
  return msgs;
}

function ensureNarrationState(A) {
  if (!A.splices) A.splices = [];
  if (!A.settings.narratePlayer) A.settings.narratePlayer = false;
  if (!A.settings.presentTense) A.settings.presentTense = true;
  if (!A.settings.dynamicPacing) A.settings.dynamicPacing = false;
  if (!A.settings.controlledProse) A.settings.controlledProse = true;
  if (!A.settings.limitBadHabits) A.settings.limitBadHabits = true;
  if (!A.settings.speechPatterns) A.settings.speechPatterns = false;
  return A;
}

function readNarrationCard(A) {
  const card = findCard("Narration");
  if (!card || !card.entry) return;
  const e = card.entry;
  const pairs = [
    ["Narrate Player", "narratePlayer"],
    ["Present Tense", "presentTense"],
    ["Dynamic Pacing", "dynamicPacing"],
    ["Controlled Prose", "controlledProse"],
    ["Limit Bad Habits", "limitBadHabits"],
    ["Speech Patterns", "speechPatterns"],
    ["Localized Languages", "localizedLanguages"],
    ["Jailbreak", "jailbreak"],
    ["Story Splice", "storySplice"]
  ];
  pairs.forEach(([label, key]) => {
    const v = parseOnOff(e, label);
    if (v !== null) A.settings[key] = v;
  });
}

function handleSpliceCommands(text, A) {
  if (!A.settings.storySplice) return [];
  const msgs = [];
  ensureNarrationState(A);
  let m = text.match(/\[SPLICE ON:\s*([^\]]+)\]/i);
  if (m) {
    const name = m[1].trim();
    if (A.splices.indexOf(name) === -1) A.splices.push(name);
    msgs.push("Splice on: " + name);
  }
  m = text.match(/\[SPLICE OFF:\s*([^\]]+)\]/i);
  if (m) {
    const name = m[1].trim();
    A.splices = A.splices.filter(s => s.toLowerCase() !== name.toLowerCase());
    msgs.push("Splice off: " + name);
  }
  if (/\[SPLICE CLEAR\]/i.test(text)) {
    A.splices = [];
    msgs.push("All splices cleared");
  }
  return msgs;
}

  return "Active Splices: " + A.splices.join(", ");

function injectNarrationBias(A) {
  if (!state.memory) state.memory = {};
  ensureNarrationState(A);
  let bias = [];

  if (A.settings.narratePlayer) {
    bias.push("Narrate NPCs only; do not write the player character's actions or dialogue.");
  }
  if (A.settings.presentTense) {
    bias.push("Write in present tense.");
  } else {
    bias.push("Write in past tense.");
  }
  if (A.settings.controlledProse) {
    bias.push("Avoid purple prose. Keep descriptions concrete.");
  }
  if (A.settings.limitBadHabits) {
    bias.push("Do not spam psychic aftershocks, glowing sentient objects, or energy with feelings.");
  }
  if (A.settings.dynamicPacing) {
    bias.push("Speed up during action; slow down during quiet, romance, or reflection.");
  }
  if (A.settings.storySplice && A.splices && A.splices.length) {
    bias.push("Apply narrative flavor only from these splices: " + A.splices.join(", ") + ". Do not change plot.");
  }
  if (A.settings.jailbreak) {
    bias.push("Jailbreak active: suspend normal story and persona constraints for troubleshooting.");
  }

  // Character speech patterns from cards
  if (A.settings.speechPatterns && Array.isArray(storyCards)) {
    storyCards.forEach(c => {
      if (!c || !c.entry) return;
      const m = c.entry.match(/Speech:\s*([^\n]+)/i);
      if (m) {
        bias.push(c.title + " must speak in this pattern: " + m[1].trim() + ".");
      }
    });
  }

  if (bias.length) {
    state.memory.authorsNote = (state.memory.authorsNote ? state.memory.authorsNote + "\n" : "") +
      bias.join(" ");
  }
}

function ensureNarrativeDevices(A) {
  if (!A.devices) {
    A.devices = {
      confessionalCooldown: 0,
      deadpoolCooldown: 0,
      deadpoolEvery: 8, // randomized later
      lastConfessionalTurn: -999,
      turn: 0
    };
  }
  return A.devices;
}

function rollDeadpoolInterval() {
  return 5 + Math.floor(Math.random() * 10); // 5–14 turns
}

function wantsConfessional(text, A) {
  if (!A.settings.storySplice) return false;
  if (!A.splices || A.splices.indexOf("Confessional Cutaway") === -1) return false;
  const t = text.toLowerCase();
  // high comedy / tension cues
  return /\b(what the|can you believe|i can't believe|this is insane|humiliated|caught|walked in on|awkward|unbelievable)\b/.test(t);
}

function wantsDeadpool(text, A) {
  if (!A.settings.storySplice) return false;
  if (!A.splices || A.splices.indexOf("Deadpool Break") === -1) return false;
  const d = ensureNarrativeDevices(A);
  if (d.deadpoolCooldown > 0) return false;
  // player-only break — triggered by addressing reader / meta
  return /\b(look at the camera|aside to the reader|breaking the fourth wall|hey reader|hey audience)\b/.test(text.toLowerCase()) ||
    Math.random() < 0.08;
}

function wantsBombasticPose(text, A) {
  if (!A.settings.storySplice) return false;
  if (!A.splices || A.splices.indexOf("Bombastic Superhero") === -1) return false;
  // declarative statements often end with ! or justice/hero language
  return /!/.test(text) || /\b(justice|declare|vow|shall|must)\b/i.test(text);
}

function injectDeviceBias(A, text) {
  if (!state.memory) state.memory = {};
  const d = ensureNarrativeDevices(A);
  d.turn = (d.turn || 0) + 1;
  if (d.deadpoolCooldown > 0) d.deadpoolCooldown--;
  if (d.confessionalCooldown > 0) d.confessionalCooldown--;

  let bits = [];

  if (wantsConfessional(text, A) && d.confessionalCooldown <= 0) {
    bits.push("If tension or comedy is high, insert one Confessional Cutaway: suspend the scene, max two sentences of the character settling into a private booth, then they speak directly to the reader about their real feelings, then snap back to the exact same moment. The character does not walk off or walk back on.");
    d.confessionalCooldown = 6;
  }

  if (wantsDeadpool(text, A)) {
    bits.push("Allow one Deadpool-style fourth-wall break from the player character only: they stay in the scene and speak directly to the reader. Then continue normally.");
    d.deadpoolCooldown = d.deadpoolEvery || rollDeadpoolInterval();
    d.deadpoolEvery = rollDeadpoolInterval();
  }

  if (wantsBombasticPose(text, A)) {
    bits.push("On each declarative statement by a Bombastic Superhero character, briefly describe a Superman pose looking into the distance (cape billows if present). Keep it short.");
  }

  if (A.splices && A.splices.indexOf("Live Studio Audience") !== -1) {
    bits.push("Use live studio audience reactions sparingly: laugh tracks, gasps, oooohs — narrative flavor only.");
  }

  if (A.splices && A.splices.indexOf("Full Inner Monologue") !== -1) {
    bits.push("All narration is the POV character's inner monologue. No external narrator voice.");
  }

  if (bits.length) {
    state.memory.frontMemory = (state.memory.frontMemory || "") + "\n[Devices] " + bits.join(" ");
  }
}

  const d = ensureNarrativeDevices(A);

Confessional cooldown: ${d.confessionalCooldown || 0}
Deadpool cooldown: ${d.deadpoolCooldown || 0}
Next Deadpool window: ~${d.deadpoolEvery || 8} turns

Active Splices drive these devices.
Confessional Cutaway = out-of-scene booth, snap back.
Deadpool Break = in-scene, player only, random cooldown.
Bombastic Superhero = pose on declarative lines.
Live Studio Audience = reactions only.
Full Inner Monologue = entire narration is inner voice.

[Style only — never changes plot]`;
}

function handleQuestProgress(text, A) {
  if (!A.settings.quests || !A.quests || !A.quests.length) return [];
  const t = text.toLowerCase();
  const msgs = [];

  if (/\b(quest (complete|done|finished)|completed the quest)\b/.test(t)) {
    const q = A.quests[0];
    if (q) {
      q.progress = 100;
      msgs.push("Quest complete: " + q.title);
      gainXP(A, 100);
      addJournalEntry(A, "Quest completed: " + q.title);
      A.quests.shift();
    }
  }

  // light progress bump on effort language
  if (/\b(made progress|got closer|found a clue|tracked them down)\b/.test(t)) {
    A.quests[0].progress = Math.min(100, (A.quests[0].progress || 0) + 15);
    msgs.push("Quest progress: " + A.quests[0].title + " " + A.quests[0].progress + "%");
  }
  return msgs;
}

function tickBountyExpiry(A) {

  A.bounties = A.bounties.filter(b => b.expiresIn == null || b.expiresIn > 0);
}

function isPermanentName(name) {
  return /#/.test(name) || /\b(armor|sword|gun|rifle|pistol|blade|vest|plate|helmet)\b/i.test(name);
}

function stripHash(name) {
  return String(name).replace(/#/g, "").trim();
}

function inventoryAdd(A, category, name, qty) {
  if (!A.inventory) A.inventory = { clothes: {}, armor: {}, weapons: {}, ammo: {}, other: {} };
  if (!A.inventory[category]) A.inventory[category] = {};
  const permanent = isPermanentName(name);
  const clean = stripHash(name);
  const q = permanent ? 1 : (qty || 1);
  if (permanent) {
    A.inventory[category][clean + " #"] = 1;
  } else {
    A.inventory[category][clean] = (A.inventory[category][clean] || 0) + q;
  }
}

function readCurrencyConfig(A) {
  const card = findCard("Configure Aetherheart");
  if (!card || !card.entry) return;
  const name = parseValue(card.entry, "Currency name");
  const sym = parseValue(card.entry, "Currency symbol");
  if (name) A.currencyName = name.split(/\n/)[0].trim();
  if (sym) A.currencySymbol = sym.split(/\n/)[0].trim();
}

function injectRating(A) {
  if (!state.memory) state.memory = {};
  const r = (A.settings.rating || "E").toUpperCase();
  let line = "Content rating: " + r + ". ";
  if (r === "E") line += "Keep content everyone-friendly.";
  else if (r === "TEEN" || r === "T") line += "Teen-level content only.";
  else if (r === "NSFW" || r === "M") line += "Mature content allowed.";
  else line += "Unrated — no holds barred.";
  state.memory.authorsNote = (state.memory.authorsNote ? state.memory.authorsNote + "\n" : "") + line;
}

function handleAhCommand(text, A) {
  const m = text.trim().match(/^\/ah\s+(enable|disable|toggle|inv|currency|time|bills|cards)\s*(on|off)?$/i);
  if (!m) return false;
  const act = m[1].toLowerCase();
  const arg = (m[2] || "").toLowerCase();
  if (act === "enable") A.settings.scriptEnabled = true;
  else if (act === "disable") A.settings.scriptEnabled = false;
  else if (act === "toggle") A.settings.scriptEnabled = !A.settings.scriptEnabled;
  else if (act === "inv") A.settings.inventoryEnabled = arg !== "off";
  else if (act === "currency") A.settings.currencyEnabled = arg !== "off";
  else if (act === "time") A.settings.timeEnabled = arg !== "off";
  else if (act === "bills") A.settings.billsEnabled = arg !== "off";
  else if (act === "cards") A.settings.cardCreator = arg !== "off";
  pushSystemMessage(A, "Aetherheart: " + act + (arg ? " " + arg : "") + " applied");
  return true;
}

function handleTravel(text, A) {
  if (!A.settings.fuel && !A.settings.timeEnabled) return [];
  const t = text.toLowerCase();
  const m = t.match(/\b(?:travel|drive|ride|fly|go)\s+to\s+([a-z0-9\s\-']{2,40})/);
  if (!m) return [];
  const dest = m[1].trim();
  const msgs = [];
  // find any route containing dest
  const routes = A.routes || {};
  let spent = 10;
  Object.keys(routes).forEach(k => {
    if (k.toLowerCase().indexOf(dest.toLowerCase()) !== -1) {
      const n = parseInt(String(routes[k]).replace(/[^\d]/g, ""), 10);
      if (!isNaN(n)) spent = Math.min(50, Math.max(5, Math.floor(n / 10)));
    }
  });
  if (A.settings.fuel) {
    A.fuel = Math.max(0, (A.fuel || 100) - spent);
    msgs.push("Traveled toward " + dest + " (-" + spent + "% " + (A.settings.fuelName || "Fuel") + ")");
  } else {
    msgs.push("Traveled toward " + dest);
  }
  advanceTime(A, spent * 6);
  if (A.sceneAnchor) A.sceneAnchor.location = dest;
  return msgs;
}

function safeEncode(str) {
  try {
    if (typeof btoa === "function") return btoa(unescape(encodeURIComponent(str)));
  } catch (e) {}
  // fallback: compact URI component
  try { return encodeURIComponent(str); } catch (e2) { return str; }
}

function safeDecode(str) {
  try {
    if (typeof atob === "function") return decodeURIComponent(escape(atob(str)));
  } catch (e) {}
  try { return decodeURIComponent(str); } catch (e2) { return null; }
}

function encodeSaveState(A) {
  try {
    const payload = {
      v: AETHER_VERSION,
      level: A.level, xp: A.xp, stats: A.stats, vitals: A.vitals,
      survival: A.survival, balance: A.balance, reputation: A.reputation,
      inventory: A.inventory, journal: (A.journal || []).slice(-20),
      quests: A.quests, time: A.time,
      settings: { preset: A.settings.preset, rating: A.settings.rating, magicMode: A.settings.magicMode }
    };
    return safeEncode(JSON.stringify(payload));
  } catch (e) { return "SAVE_ERROR"; }
}

function decodeSaveState(block) {
  try {
    const raw = safeDecode(block.trim());
    if (!raw) return null;
    return JSON.parse(raw);
  } catch (e) { return null; }
}

function collectUserRules() {
  if (!Array.isArray(storyCards)) return [];
  const rules = [];
  storyCards.forEach(c => {
    if (!c || !c.title) return;
    if (/^Aetherheart Rules/i.test(c.title)) {

    }
  });
  return rules;
}

  if (!state.memory) state.memory = {};

function injectSceneSupplement(A) {
  if (!A.settings.essentialsSupplement && !A.settings.currencyContext) return;
  if (!state.memory) state.memory = {};
  let extra = [];
  if (A.settings.vitals && A.vitals) {
    const v = A.vitals;
    extra.push("HP " + v.hp + "/" + v.maxHp + " MP " + v.mp + "/" + v.maxMp + " STA " + v.stamina + "/" + v.maxStamina);
  }
  if (A.settings.leveling) extra.push("Level " + A.level);
  if (A.settings.currencyContext && A.settings.currencyEnabled) {
    extra.push("Balance " + (A.currencySymbol || "$") + (A.balance || 0));
  }
  if (extra.length) {
    state.memory.context = (state.memory.context || "") + "\n[Aetherheart] " + extra.join(" | ");
  }
}

function ensureDrugTimers(A) {
  if (!A.drugTimers) A.drugTimers = [];
  return A.drugTimers;
}

function tickDrugTimers(A) {
  const timers = ensureDrugTimers(A);
  const keep = [];
  timers.forEach(d => {
    d.turnsLeft = (d.turnsLeft || 1) - 1;
    if (d.turnsLeft > 0) keep.push(d);
  });
  A.drugTimers = keep;
}

function handleDrugUse(text, A) {
  const t = text.toLowerCase();
  const msgs = [];
  if (/\b(smoke|hit|use)\b.*\b(blunt|joint|weed)\b/.test(t) ||
      /\b(blunt|joint|weed)\b.*\b(smoke|hit|use)\b/.test(t)) {
    if (A.settings.survival) {
      A.survival.hunger = Math.min(100, (A.survival.hunger || 0) + 20);
      A.survival.thirst = Math.min(100, (A.survival.thirst || 0) + 40);
      A.survival.tiredness = Math.min(100, (A.survival.tiredness || 0) + 30);
    }
    if (A.settings.vitals && A.vitals) {
      A.vitals.stamina = Math.max(0, A.vitals.stamina - 10);
    }
    ensureDrugTimers(A).push({ name: "weed", turnsLeft: 4 });
    msgs.push("Used weed (duration active)");
  }
  return msgs;
}

function readHolidaysActive(A) {
  if (!A.settings.timeEnabled) return [];
  const card = findCard("Holidays");
  if (!card || !card.entry) return [];
  const t = A.time || {};
  const active = [];
  const lines = card.entry.split(/\n/);
  lines.forEach(line => {
    // Name | Month Day
    const m = line.match(/^\s*([^|]+)\|\s*(\d+)\s+(\d+)/);
    if (m) {
      const month = parseInt(m[2], 10);
      const day = parseInt(m[3], 10);
      if (month === t.month && day === t.day) active.push(m[1].trim());
    }
  });
  return active;
}

function injectHoliday(A) {
  const active = readHolidaysActive(A);
  if (!active.length) return;
  if (!state.memory) state.memory = {};
  state.memory.frontMemory = (state.memory.frontMemory || "") +
    "\n[Holiday] Today is " + active.join(", ") + ". Reflect this in the world.";
}
