// ===================================================================
// AETHERHEART — OUTPUT
// Paste this whole file into the "Output" script tab.
// Handles turn advancement, survival ticks, rest/refuel actions, and post-processing.
// Minimal: preserves Aetherheart patterns and avoids echoing the hidden world state.
// ===================================================================

function onOutput_AETHER(text) {
  try {
    ahInit();
    var s = state.aetherheart;
    var f = s.flags = s.flags || {};

    var prefix = "";

    // Handle d20 roll result prefix
    if (f.d20Roll) {
      prefix += "🎲 d20 " + (f.d20Roll.kind || "") + " roll: " + f.d20Roll.roll + "\n\n";
      f.d20Roll = null;
    }

    // Rest handling: if a rest flag was set during input processing, apply rest logic
    if (f.rest) {
      ahRest();
      prefix += "(Rested. Time passed.)\n\n";
      f.rest = false;
    } else {
      // Normal turn time advance
      var mpt = (s.settings && s.settings.minutesPerTurn) ? s.settings.minutesPerTurn : 120;
      ahAdvanceTime(mpt);
    }

    // Tick survival & tickers
    if (s.settings && s.settings.survival) ahTickSurvival((s.settings && s.settings.minutesPerTurn) ? s.settings.minutesPerTurn : 120);
    if (s.tickers) ahTickTickers();

    // Card creator processing
    text = ahRunCardCreator(text);

    // Attempt to strip any accidental model echo of the world-state block (best-effort)
    try {
      var ws = ahBuildWorldState();
      if (text.indexOf(ws) === 0) {
        // If model repeated the block verbatim, strip it out
        text = text.slice(ws.length).trim();
      }
    } catch (e) { /* ignore */ }

    return prefix + text;
  } catch (e) {
    try { state.message = "AETHER Output: " + (e && e.message ? e.message : e); } catch (e2) {}
    return text;
  }
}

const modifier = (text) => {
  try {
    text = onOutput_AETHER(text);
  } catch (e) {
    try { state.message = "AETHER Output: " + (e && e.message ? e.message : e); } catch (e2) {}
  }
  return { text };
};
modifier(text);
