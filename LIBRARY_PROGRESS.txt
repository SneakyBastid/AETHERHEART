// LIBRARY - AETHERHEART - CHARACTER SHEET & JOURNAL MODULE
// Adds Character progression (XP, level, stats, skills, perks, powers) and Journal auto-summaries

(function(){
  var STATE_ROOT = 'aetherheart';

  function ensureState() {
    if (typeof state !== 'object') return null;
    state[STATE_ROOT] = state[STATE_ROOT] || {};
    state[STATE_ROOT].character = state[STATE_ROOT].character || {
      level: 1,
      xp: 0,
      xpToNext: 100,
      stats: { STR: 10, DEX: 10, CON: 10, INT: 10, WIS: 10, CHA: 10 },
      skills: [],
      perks: [],
      powers: []
    };
    state[STATE_ROOT].journal = state[STATE_ROOT].journal || {
      entries: [],
      autoSummaryTurns: 8,
      lastSummaryTurn: 0,
      summaries: []
    };
    return state[STATE_ROOT];
  }

  // ------------------ Character progression ------------------
  function getCharacter() { var s = ensureState(); if (!s) return null; return s.character; }

  function xpForNext(level) {
    // simple escalating requirement: base 100 * level^1.25
    return Math.max(50, Math.round(100 * Math.pow(level, 1.25)));
  }

  function addXp(amount) {
    var s = ensureState(); if (!s) return null;
    var c = s.character;
    c.xp = (c.xp || 0) + Math.max(0, Math.floor(amount));
    // level up loop
    var leveled = [];
    while (c.xp >= (c.xpToNext || xpForNext(c.level))) {
      var needed = c.xpToNext || xpForNext(c.level);
      c.xp -= needed;
      c.level = (c.level || 1) + 1;
      c.xpToNext = xpForNext(c.level);
      // simple reward on level-up: add a power and bump a stat
      c.powers = c.powers || [];
      c.powers.push('LevelPower_' + c.level);
      // bump a stat by 1 (round-robin)
      var statsKeys = Object.keys(c.stats || {});
      var idx = (c.level - 1) % statsKeys.length;
      if (statsKeys[idx]) c.stats[statsKeys[idx]] = (c.stats[statsKeys[idx]] || 0) + 1;
      leveled.push({ level: c.level });
    }
    return { xp: c.xp, level: c.level, leveled: leveled };
  }

  function characterStatusText() {
    var c = getCharacter(); if (!c) return 'Character data unavailable.';
    var out = [];
    out.push('Level: ' + c.level + '   XP: ' + (c.xp || 0) + ' / ' + (c.xpToNext || xpForNext(c.level)));
    out.push('\n-- STATS --');
    var stats = [];
    for (var k in c.stats) { if (c.stats.hasOwnProperty(k)) stats.push(k + ' ' + c.stats[k]); }
    out.push(stats.join('   '));
    out.push('\n-- SKILLS --');
    out.push((c.skills && c.skills.length) ? c.skills.join(', ') : '(none)');
    out.push('\n-- PERKS --');
    out.push((c.perks && c.perks.length) ? c.perks.join(', ') : '(none)');
    out.push('\n-- POWERS --');
    out.push((c.powers && c.powers.length) ? c.powers.join(', ') : '(none)');
    return out.join('\n');
  }

  // onInput handler for /sheet and /xp commands
  function handleCharacterCommands(text) {
    if (!text) return { text: text };
    var t = text.trim();
    if (t.indexOf('/sheet') === 0) {
      return { text: characterStatusText() };
    }
    var m = t.match(/^\/xp\s+(\d+)/i);
    if (m) {
      var amt = parseInt(m[1],10);
      var res = addXp(amt);
      return { text: 'Gained ' + amt + ' XP. Level: ' + res.level + '. Current XP: ' + res.xp };
    }
    return { text: text };
  }

  // expose API
  var Character = {
    get: getCharacter,
    addXp: addXp,
    statusText: characterStatusText
  };
  if (typeof window !== 'undefined') { window.AetherheartCharacter = window.AetherheartCharacter || Character; } else { try { this.AetherheartCharacter = this.AetherheartCharacter || Character; } catch (e) {} }

  // ------------------ Journal & Auto-summaries ------------------
  function addJournalEntry(text) {
    var s = ensureState(); if (!s) return false;
    var j = s.journal;
    var entry = { time: Date.now(), text: (text||'').trim() };
    j.entries.push(entry);
    return entry;
  }

  function lastNEntries(n) {
    var s = ensureState(); if (!s) return [];
    var e = s.journal.entries || [];
    return e.slice(Math.max(0, e.length - n));
  }

  function generateSummaryFromEntries(entries) {
    // Lightweight summary: join entries and truncate to 400 chars; preserve sentence boundaries if possible.
    var joined = entries.map(function(en){ return en.text; }).join(' ');
    // naive sentence split
    var sentences = joined.match(/[^\.\!\?]+[\.\!\?]?/g) || [joined];
    var out = '';
    for (var i = 0; i < sentences.length; i++) {
      if ((out + sentences[i]).length > 400) break;
      out += sentences[i].trim() + ' ';
    }
    return out.trim();
  }

  function forceJournalSummary() {
    var s = ensureState(); if (!s) return null;
    var j = s.journal;
    var entries = lastNEntries(20);
    if (!entries.length) return null;
    var summary = generateSummaryFromEntries(entries);
    var note = { createdAt: Date.now(), summary: summary };
    j.summaries = j.summaries || [];
    j.summaries.push(note);
    j.lastSummaryTurn = (j.lastSummaryTurn || 0) + 1; // placeholder for turn tracking
    return note;
  }

  function journalStatusText() {
    var s = ensureState(); if (!s) return 'Journal unavailable.';
    var j = s.journal;
    var out = [];
    out.push('Entries: ' + (j.entries ? j.entries.length : 0));
    out.push('Next auto-summary in: ' + (j.autoSummaryTurns || 8) + ' turns');
    out.push('\nLast summaries:');
    var sums = j.summaries || [];
    if (!sums.length) out.push('(none)'); else out.push(sums.slice(-3).map(function(su){ return '- ' + (new Date(su.createdAt)).toISOString().split('T')[0] + ': ' + (su.summary.length > 140 ? su.summary.slice(0,140) + '...' : su.summary); }).join('\n'));
    return out.join('\n');
  }

  function handleJournalCommands(text) {
    if (!text) return { text: text };
    var t = text.trim();
    if (t.indexOf('/journal') === 0) {
      var mAdd = t.match(/^\/journal\s+add\s+(.+)$/i);
      if (mAdd) {
        var entry = addJournalEntry(mAdd[1]);
        return { text: 'Journal entry added.' };
      }
      if (/^\/journal\s+force-summary/i.test(t)) {
        var s = forceJournalSummary();
        if (!s) return { text: 'Nothing to summarize.' };
        return { text: 'Forced summary created.' };
      }
      return { text: journalStatusText() };
    }
    return { text: text };
  }

  // Expose API
  var Journal = {
    add: addJournalEntry,
    forceSummary: forceJournalSummary,
    statusText: journalStatusText
  };
  if (typeof window !== 'undefined') { window.AetherheartJournal = window.AetherheartJournal || Journal; } else { try { this.AetherheartJournal = this.AetherheartJournal || Journal; } catch (e) {} }

  // Integrate with modifier pipeline
  function onInput_progress(text) {
    if (!text) return { text: text };
    var r = handleCharacterCommands(text);
    if (r && typeof r.text === 'string' && r.text !== text) return { text: r.text };
    var r2 = handleJournalCommands(text);
    if (r2 && typeof r2.text === 'string' && r2.text !== text) return { text: r2.text };
    return { text: text };
  }

  if (typeof window !== 'undefined') {
    window.onInput_progress = window.onInput_progress || onInput_progress;
  } else {
    try { this.onInput_progress = this.onInput_progress || onInput_progress; } catch (e) {}
  }

  try {
    if (typeof modifier === 'function' && modifier.__patched_for_progress !== true) {
      var original = modifier;
      var patched = function(text){
        var r = onInput_progress(text);
        if (r && typeof r.text === 'string' && r.text !== text) return r.text;
        return original(text);
      };
      patched.__patched_for_progress = true;
      try { modifier = patched; } catch (e) {}
    }
  } catch (e) {}

})();
